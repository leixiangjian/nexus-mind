1. 数据库概览

1.1 数据库实例

项目配置
数据库引擎PostgreSQL 16
扩展uuid-ossp, pgvector, pg_trgm
字符集UTF8
排序规则en_US.UTF-8
连接池asyncpg + SQLAlchemy (pool_size=10, max_overflow=20)
主键策略UUID v4（gen_random_uuid()）
时间字段TIMESTAMPTZ，统一 UTC
软删除不使用，物理删除 + 外键级联

3. 完整 DDL 脚本

-- ============================================================
-- NexusMind 数据库初始化脚本
-- PostgreSQL 16 + pgvector
-- ============================================================

CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
CREATE EXTENSION IF NOT EXISTS "pgvector";
CREATE EXTENSION IF NOT EXISTS "pg_trgm";

-- 1. 租户
CREATE TABLE tenants (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name            VARCHAR(128) NOT NULL,
    slug            VARCHAR(64) UNIQUE NOT NULL,
    plan            VARCHAR(32) DEFAULT 'free',
    settings        JSONB DEFAULT '{}',
    is_active       BOOLEAN DEFAULT TRUE,
    created_at      TIMESTAMPTZ DEFAULT NOW(),
    updated_at      TIMESTAMPTZ DEFAULT NOW()
);

-- 2. 用户
CREATE TABLE users (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id       UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    email           VARCHAR(255) NOT NULL,
    hashed_password VARCHAR(255) NOT NULL,
    display_name    VARCHAR(128),
    role            VARCHAR(32) DEFAULT 'member',
    is_active       BOOLEAN DEFAULT TRUE,
    last_login_at   TIMESTAMPTZ,
    created_at      TIMESTAMPTZ DEFAULT NOW(),
    updated_at      TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(tenant_id, email)
);
CREATE INDEX idx_users_tenant ON users(tenant_id);

-- 3. 大模型配置
CREATE TABLE model_configs (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id       UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    name            VARCHAR(128) NOT NULL,
    provider        VARCHAR(64) NOT NULL,
    model_id        VARCHAR(128) NOT NULL,
    api_key_enc     BYTEA NOT NULL,
    base_url        VARCHAR(512),
    temperature     NUMERIC(3,2) DEFAULT 0.7,
    max_tokens      INTEGER DEFAULT 8192,
    context_length  INTEGER DEFAULT 128000,
    top_p           NUMERIC(3,2) DEFAULT 1.0,
    extra_config    JSONB DEFAULT '{}',
    is_enabled      BOOLEAN DEFAULT FALSE,
    health_status   VARCHAR(32) DEFAULT 'unknown',
    last_tested_at  TIMESTAMPTZ,
    sort_order      INTEGER DEFAULT 0,
    created_at      TIMESTAMPTZ DEFAULT NOW(),
    updated_at      TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(tenant_id, name)
);
CREATE INDEX idx_model_configs_tenant ON model_configs(tenant_id);

-- 4. 虚拟系统
CREATE TABLE virtual_systems (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id       UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    name            VARCHAR(128) NOT NULL,
    hostname        VARCHAR(128) NOT NULL,
    ip_address      VARCHAR(45),
    os_image        VARCHAR(128) DEFAULT 'ubuntu:22.04',
    vcpu            INTEGER DEFAULT 2,
    memory_mb       INTEGER DEFAULT 4096,
    disk_mb         INTEGER DEFAULT 20480,
    status          VARCHAR(32) DEFAULT 'ready',
    container_id    VARCHAR(128),
    sandbox_config  JSONB DEFAULT '{}',
    created_at      TIMESTAMPTZ DEFAULT NOW(),
    updated_at      TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(tenant_id, name)
);
CREATE INDEX idx_virtual_systems_tenant ON virtual_systems(tenant_id);

-- 5. 网络规则
CREATE TABLE vm_network_rules (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    vm_id           UUID NOT NULL REFERENCES virtual_systems(id) ON DELETE CASCADE,
    host_pattern    VARCHAR(255) NOT NULL,
    port            INTEGER DEFAULT 0,
    is_allowed      BOOLEAN DEFAULT TRUE,
    description     VARCHAR(255),
    created_at      TIMESTAMPTZ DEFAULT NOW()
);
CREATE INDEX idx_vm_network_rules_vm ON vm_network_rules(vm_id);

-- 6. 文件规则
CREATE TABLE vm_file_rules (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    vm_id           UUID NOT NULL REFERENCES virtual_systems(id) ON DELETE CASCADE,
    path_pattern    VARCHAR(512) NOT NULL,
    permission      VARCHAR(16) DEFAULT 'rw',
    description     VARCHAR(255),
    created_at      TIMESTAMPTZ DEFAULT NOW()
);
CREATE INDEX idx_vm_file_rules_vm ON vm_file_rules(vm_id);

-- 7. 角色
CREATE TABLE roles (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id       UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    name            VARCHAR(128) NOT NULL,
    icon            VARCHAR(16) DEFAULT '✦',
    color           VARCHAR(16) DEFAULT '#8b5cf6',
    ability         VARCHAR(128),
    description     TEXT,
    vm_id           UUID REFERENCES virtual_systems(id) ON DELETE SET NULL,
    system_prompt   TEXT,
    created_at      TIMESTAMPTZ DEFAULT NOW(),
    updated_at      TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(tenant_id, name)
);
CREATE INDEX idx_roles_tenant ON roles(tenant_id);
CREATE INDEX idx_roles_vm ON roles(vm_id);

-- 8. 机器人
CREATE TABLE robots (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id       UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    name            VARCHAR(128) NOT NULL,
    role_id         UUID NOT NULL REFERENCES roles(id) ON DELETE RESTRICT,
    color           VARCHAR(16),
    status          VARCHAR(32) DEFAULT 'idle',
    total_tasks     INTEGER DEFAULT 0,
    success_rate    NUMERIC(5,2) DEFAULT 100.0,
    current_exec_id UUID,
    note            TEXT,
    created_at      TIMESTAMPTZ DEFAULT NOW(),
    updated_at      TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(tenant_id, name)
);
CREATE INDEX idx_robots_tenant ON robots(tenant_id);
CREATE INDEX idx_robots_role ON robots(role_id);
CREATE INDEX idx_robots_status ON robots(tenant_id, status);

-- 9. 任务
CREATE TABLE tasks (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id       UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    name            VARCHAR(256) NOT NULL,
    agent_type      VARCHAR(32) NOT NULL,
    status          VARCHAR(32) DEFAULT 'draft',
    role_id         UUID REFERENCES roles(id) ON DELETE SET NULL,
    model_config_id UUID REFERENCES model_configs(id) ON DELETE SET NULL,
    icon            VARCHAR(16) DEFAULT '✦',
    color           VARCHAR(16) DEFAULT '#8b5cf6',
    prompt          TEXT,
    system_prompt   TEXT,
    timeout_sec     INTEGER DEFAULT 120,
    max_retries     INTEGER DEFAULT 2,
    output_format   VARCHAR(32) DEFAULT 'json',
    extra_config    JSONB DEFAULT '{}',
    last_tested_at  TIMESTAMPTZ,
    version         INTEGER DEFAULT 1,
    created_at      TIMESTAMPTZ DEFAULT NOW(),
    updated_at      TIMESTAMPTZ DEFAULT NOW()
);
CREATE INDEX idx_tasks_tenant ON tasks(tenant_id);
CREATE INDEX idx_tasks_status ON tasks(tenant_id, status);

-- 10. 测试用例
CREATE TABLE test_cases (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    task_id         UUID NOT NULL REFERENCES tasks(id) ON DELETE CASCADE,
    name            VARCHAR(256) NOT NULL,
    input_desc      TEXT,
    input_data      JSONB,
    expected_output TEXT,

5. 事务与并发控制

5.1 事务隔离级别

场景隔离级别说明
常规读写READ COMMITTEDPostgreSQL 默认，性能好
智能分配REPEATABLE READ避免幻读导致重复分配
并发机器人分配行级锁 + SKIP LOCKEDSELECT ... FOR UPDATE SKIP LOCKED

5.2 机器人分配锁

BEGIN;
SELECT id, name, role_id FROM robots
WHERE tenant_id = $1
  AND status = 'idle'
  AND role_id = $2
ORDER BY success_rate DESC
LIMIT 1
FOR UPDATE SKIP LOCKED;

UPDATE robots SET status = 'busy', current_exec_id = $3 WHERE id = $4;
COMMIT;
