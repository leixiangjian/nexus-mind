    builder.add_node("load_work", load_work)
    builder.add_node("validate_dag", validate_dag)
    builder.add_node("topological_sort", topological_sort)
    builder.add_node("analyze_dependencies", analyze_dependencies)

    builder.add_edge(START, "load_work")

    builder.add_edge("analyze_dependencies", "build_plan")
    builder.add_edge("build_plan", "report")
    builder.add_edge("report", END)

    return builder.compile(checkpointer=checkpointer)

async def load_work(state: WorkflowState) -> dict:
    """加载工作及其节点、边"""
    work_id = state["work_id"]

    async with async_session_factory() as db:
        work = await db.get(Work, work_id)
        if not work:
            return {"validation_errors": [f"Work {work_id} not found"]}

        nodes_query = select(WorkNode).where(WorkNode.work_id == work_id)
        edges_query = select(WorkEdge).where(WorkEdge.work_id == work_id)
        nodes = (await db.execute(nodes_query)).scalars().all()
        edges = (await db.execute(edges_query)).scalars().all()

        node_states: dict[str, dict] = {}
        for n in nodes:
            task = await db.get(Task, n.task_id)
            node_states[str(n.id)] = {
                "node_id": str(n.id),
                "task_id": str(n.task_id),
                "task_name": task.name if task else "未知任务",
                "node_type": n.node_type,
                "role_id": str(task.role_id) if task and task.role_id else None,
                "status": "pending",
                "robot_id": None,
                "robot_name": None,
                "output": None,
                "error": None,
                "started_at": None,
                "finished_at": None,
                "retry_count": 0,
            }

        edge_list = [
            (str(e.from_node_id), str(e.to_node_id), e.edge_type)
            for e in edges
        ]

    return {
        "work_name": work.name,
        "nodes": node_states,
        "edges": edge_list,
    }


async def validate_dag(state: WorkflowState) -> dict:
    """验证 DAG 结构：无环、无孤立节点、有开始和结束"""
    errors: list[str] = []
    nodes = state["nodes"]
    edges = state["edges"]

    if not nodes:
        errors.append("工作没有任何节点")
        return {"validation_errors": errors}

    # 环检测（Kahn 算法）
    indeg: dict[str, int] = {nid: 0 for nid in nodes}
    adj: dict[str, list[str]] = defaultdict(list)
    for from_id, to_id, _ in edges:
        adj[from_id].append(to_id)
        indeg[to_id] = indeg.get(to_id, 0) + 1

    queue = [nid for nid, d in indeg.items() if d == 0]
    visited = 0
    while queue:
        cur = queue.pop()
        visited += 1
        for nxt in adj[cur]:
            indeg[nxt] -= 1
            if indeg[nxt] == 0:
                queue.append(nxt)

    if visited != len(nodes):
        errors.append("工作中存在循环依赖")

    # 孤立节点检查
    connected = set()
    for f, t, _ in edges:
        connected.add(f)
        connected.add(t)
    orphans = set(nodes.keys()) - connected
    if len(nodes) > 1 and orphans:
        errors.append(f"存在孤立节点: {orphans}")

    return {"validation_errors": errors, "cycle_detected": bool(errors)}


async def topological_sort(state: WorkflowState) -> dict:
    """拓扑排序，生成执行顺序"""
    nodes = state["nodes"]
    edges = state["edges"]

    indeg: dict[str, int] = {nid: 0 for nid in nodes}
    adj: dict[str, list[str]] = defaultdict(list)
    for f, t, _ in edges:
        adj[f].append(t)
        indeg[t] += 1

    queue = [nid for nid, d in indeg.items() if d == 0]
    order: list[str] = []
    while queue:
        cur = queue.pop(0)
        order.append(cur)
        for nxt in adj[cur]:
            indeg[nxt] -= 1
            if indeg[nxt] == 0:
                queue.append(nxt)

    return {"execution_order": order}


async def analyze_dependencies(state: WorkflowState) -> dict:
    """分析依赖关系，供执行时并发调度参考"""
    nodes = state["nodes"]
    edges = state["edges"]

    deps: dict[str, list[str]] = defaultdict(list)
    for f, t, _ in edges:
        deps[t].append(f)

    # 将依赖信息写回节点状态
    updated = {}
    for nid, node in nodes.items():
        updated[nid] = {**node, "dependencies": deps.get(nid, [])}

    return {"nodes": updated}


async def build_execution_plan(state: WorkflowState) -> dict:
    """构建执行计划（含并行分组）"""
    order = state["execution_order"]
    nodes = state["nodes"]

    # 简单实现：按拓扑序线性执行
    # 复杂实现：识别可并行的兄弟节点
    plan = []
    for idx, nid in enumerate(order):
        node = nodes[nid]
        plan.append({
            "step": idx + 1,
            "node_id": nid,
            "task_name": node["task_name"],
            "node_type": node["node_type"],
        })

    return {"execution_plan": plan}


async def report_result(state: WorkflowState) -> dict:
    """汇报验证结果"""
    errors = state.get("validation_errors", [])
    await event_bus.publish(f"workflow:{state['work_id']}", {
        "type": "workflow_validated",
        "work_id": state["work_id"],
        "valid": not errors,
        "errors": errors,
    })
    return {"messages": [{"role": "system", "content": f"验证完成，错误数: {len(errors)}"}]}

            # ... 实际测试逻辑

            duration = 0.8 + idx * 0.3

    async with async_session_factory() as db:

            task = await db.get(Task, state["target_id"])
            if task:
                task.status = "passed" if overall == "passed" else "failed"
                task.last_tested_at = "NOW()"
        else:
            work = await db.get(Work, state["target_id"])
            if work:
                work.test_status = overall
                work.test_report = {
                    "passed": state["passed_count"],
                    "failed": state["failed_count"],
                    "duration": state["elapsed_sec"],
                }
        await db.commit()

    await event_bus.publish(f"test:{state['target_id']}", {
        "type": "summary",
        "overall": overall,
        "passed": state["passed_count"],
        "failed": state["failed_count"],
        "total": state["total_count"],
        "elapsed": state["elapsed_sec"],
    })

    return {
        "overall_status": overall,
        "messages": [{
            "role": "system",
            "content": f"测试完成: {state['passed_count']}/{state['total_count']}",
        }],
    }

async def match_by_role(state: AllocatorState) -> dict:
    """按角色匹配：优先同角色空闲机器人"""
    assignments = dict(state["assignments"])
    idle_robots = list(state["idle_robots"])
    unmatched = []

    # 使用行级锁保证并发安全
    async with async_session_factory() as db:
        for node in state["robot_nodes"]:
            if node["node_id"] in assignments:
                continue

            role_id = node["role_id"]
            if not role_id:
                unmatched.append(node)
                continue

            # SELECT ... FOR UPDATE SKIP LOCKED
            query = (
                select(Robot)
                .where(and_(Robot.status == "idle", Robot.role_id == role_id))
                .order_by(Robot.success_rate.desc())
                .limit(1)
                .with_for_update(skip_locked=True)
            )
            result = (await db.execute(query)).scalar_one_or_none()

            if result:
                result.status = "busy"
                assignments[node["node_id"]] = {
                    "robot_id": str(result.id),
                    "robot_name": result.name,
                    "robot_color": result.color,
                    "role_id": role_id,
                }
                await db.flush()

                # 从空闲池移除
                idle_robots = [r for r in idle_robots if r["id"] != str(result.id)]

        await db.commit()

    return {
        "assignments": assignments,
        "idle_robots": idle_robots,
        "unmatched_remaining": unmatched,
        "assigned_count": len(assignments),
    }


async def fallback_assign(state: AllocatorState) -> dict:
    """兜底分配：任意空闲机器人"""
    assignments = dict(state["assignments"])
    idle_robots = list(state["idle_robots"])
    still_unmatched = []

    for node in state.get("unmatched_remaining", []):
        if not idle_robots:
            still_unmatched.append(node)
            continue
        # 取成功率最高的
        best = sorted(idle_robots, key=lambda r: -r["success_rate"])[0]
        assignments[node["node_id"]] = {
            "robot_id": best["id"],
            "robot_name": best["name"],
            "robot_color": None,
            "role_id": best["role_id"],
            "fallback": True,
        }
        idle_robots = [r for r in idle_robots if r["id"] != best["id"]]

    return {
        "assignments": assignments,
        "idle_robots": idle_robots,
        "still_unmatched": still_unmatched,
        "assigned_count": len(assignments),
    }


async def mark_pending(state: AllocatorState) -> dict:
    """标记无法分配的节点为待分配"""
    assignments = dict(state["assignments"])
    for node in state.get("still_unmatched", []):
        assignments[node["node_id"]] = {"pending": True}

    return {
        "assignments": assignments,
        "pending_count": len(state.get("still_unmatched", [])),
    }


async def persist_allocations(state: AllocatorState) -> dict:
    """持久化分配结果"""
    async with async_session_factory() as db:
        alloc = Allocation(
            work_id=state["work_id"],
            status="allocated",
            total_tasks=state["total_count"],
            assigned_count=state["assigned_count"],
            pending_count=state["pending_count"],
            auto_count=state["auto_count"],
            detail=state["assignments"],
        )
        db.add(alloc)
        await db.flush()

        for node_id, info in state["assignments"].items():
            item = AllocationItem(
                allocation_id=alloc.id,
                work_node_id=node_id,
                robot_id=info.get("robot_id"),
                status="assigned" if "robot_id" in info else "pending",
            )
            db.add(item)

        await db.commit()
        allocation_id = str(alloc.id)

    return {"allocation_id": allocation_id}


async def notify_result(state: AllocatorState) -> dict:
    """通知分配结果"""
    await event_bus.publish(f"allocation:{state['work_id']}", {
        "type": "allocation_complete",
        "work_id": state["work_id"],
        "allocation_id": state.get("allocation_id"),
        "total": state["total_count"],
        "assigned": state["assigned_count"],
        "pending": state["pending_count"],
    })
    return {"messages": [{"role": "system", "content": "分配完成"}]}

5.1 图定义

src/agents/executor/graph.py
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.postgres.aio import AsyncPostgresSaver

from .nodes import (
    init_execution, check_control, execute_next_node,
    should_continue, human_approval, finalize_execution,
)
from ..state import ExecutorState


def build_executor_graph(checkpointer: AsyncPostgresSaver):
    builder = StateGraph(ExecutorState)

    builder.add_node("init", init_execution)
    builder.add_node("check_control", check_control)
    builder.add_node("execute", execute_next_node)
    builder.add_node("approval", human_approval)
    builder.add_node("finalize", finalize_execution)

    builder.add_edge(START, "init")
    builder.add_edge("init", "check_control")

    builder.add_conditional_edges(
        "check_control",
        lambda s: (
            "finalize" if s.get("aborted") or s["current_index"] >= s["total_count"]
            else "approval" if s.get("needs_approval")
            else "execute"
        ),
        {
            "finalize": "finalize",
            "approval": "approval",
            "execute": "execute",
        },
    )

    builder.add_edge("execute", "check_control")
    builder.add_conditional_edges(
        "approval",
        lambda s: "finalize" if s.get("aborted") else "execute",
        {"finalize": "finalize", "execute": "execute"},
    )
    builder.add_edge("finalize", END)

    return builder.compile(checkpointer=checkpointer)

5.2 执行节点实现

src/agents/executor/nodes.py
import time
from datetime import datetime, timezone
from langgraph.types import interrupt

from src.db.session import async_session_factory
from src.models.execution import Execution, ExecutionEvent
from src.models.robot import Robot
from src.models.task import Task
from src.events.bus import event_bus
from ..state import ExecutorState


async def init_execution(state: ExecutorState) -> dict:
    """初始化执行"""
    async with async_session_factory() as db:
        execution = await db.get(Execution, state["execution_id"])
        execution.status = "running"
        execution.started_at = datetime.now(timezone.utc)
        await db.commit()

    await event_bus.publish(f"execution:{state['execution_id']}", {
        "type": "execution_started",
        "execution_id": state["execution_id"],
        "total": state["total_count"],
    })

    return {
        "current_index": 0,
        "done_count": 0,
        "paused": False,
        "aborted": False,
        "events": [],
        "started_at": datetime.now(timezone.utc).isoformat(),
    }


async def check_control(state: ExecutorState) -> dict:
    """检查暂停/停止控制信号"""
    if state.get("aborted"):
        return {"aborted": True}

    # 从 Redis 读取控制信号
    import redis.asyncio as aioredis
    from src.config import settings
    r = aioredis.from_url(settings.redis_url)

    paused = await r.get(f"execution:{state['execution_id']}:paused")
    aborted = await r.get(f"execution:{state['execution_id']}:aborted")

    await r.close()

    if paused and paused.decode() == "1":
        # 使用 LangGraph interrupt 暂停
        return {"paused": True}

    if aborted and aborted.decode() == "1":
        return {"aborted": True}

    return {"paused": False}


async def execute_next_node(state: ExecutorState) -> dict:
    """执行下一个节点"""
    idx = state["current_index"]
    node_id = state["execution_order"][idx]
    node = state["node_states"][node_id]
    assignment = state["allocation"].get(node_id, {})

    await event_bus.publish(f"execution:{state['execution_id']}", {
        "type": "node_start",
        "node_id": node_id,
        "robot_id": assignment.get("robot_id"),
        "robot_name": assignment.get("robot_name"),
    })

    # 记录事件到数据库
    async with async_session_factory() as db:
        evt = ExecutionEvent(
            execution_id=state["execution_id"],
            event_type="test",
            node_id=node_id,
            robot_id=assignment.get("robot_id"),
            message=f"开始执行: {node['task_name']}",
        )
        db.add(evt)
        await db.commit()

    # 执行节点
    if node["node_type"] == "robot" and assignment.get("robot_id"):
        result = await execute_robot_node(state, node_id, assignment)
    else:
        result = await execute_auto_node(state, node_id)

    # 更新状态
    updated_nodes = dict(state["node_states"])
    updated_nodes[node_id] = {
        **node,
        "status": result["status"],
        "output": result.get("output"),
        "error": result.get("error"),
        "finished_at": datetime.now(timezone.utc).isoformat(),
    }

    await event_bus.publish(f"execution:{state['execution_id']}", {
        "type": "node_complete",
        "node_id": node_id,
        "output": result.get("output"),
    })

    return {
        "node_states": updated_nodes,
        "current_index": idx + 1,
        "done_count": state["done_count"] + 1,
    }


async def execute_robot_node(state: ExecutorState, node_id: str, assignment: dict) -> dict:
    """执行机器人节点（调用子代理）"""
    from .bot_agent import create_bot_agent

    async with async_session_factory() as db:
        robot = await db.get(Robot, assignment["robot_id"])
        node = state["node_states"][node_id]
        task = await db.get(Task, node["task_id"])

        # 创建子代理
        agent = await create_bot_agent(robot, task)

        # 执行
        result = await agent.ainvoke({
            "messages": [{"role": "user", "content": task.prompt or task.name}]
        })

    output = result["messages"][-1].content if result.get("messages") else ""

    return {"status": "done", "output": output}


async def execute_auto_node(state: ExecutorState, node_id: str) -> dict:
    """执行自动节点（开始/结束/条件/等待等）"""
    node = state["node_states"][node_id]
    task_name = node["task_name"]

    # 自动节点通常是纯控制流，不需要 LLM 调用
    if "开始" in task_name or "结束" in task_name:
        return {"status": "done", "output": "ok"}

    return {"status": "done", "output": f"auto executed: {task_name}"}


async def human_approval(state: ExecutorState) -> dict:
    """人工审批：使用 interrupt 暂停"""
    node_id = state["execution_order"][state["current_index"]]

    approval = interrupt({
        "type": "human_approval_required",
        "execution_id": state["execution_id"],
        "node_id": node_id,
        "prompt": "请审批是否继续执行",
    })

    if approval.get("approved"):
        return {"current_index": state["current_index"] + 1}
    return {"aborted": True}


async def finalize_execution(state: ExecutorState) -> dict:
    """收尾：更新执行状态、释放机器人"""
    status = "stopped" if state.get("aborted") else "done"

    async with async_session_factory() as db:
        execution = await db.get(Execution, state["execution_id"])
        execution.status = status
        execution.done_nodes = state["done_count"]
        execution.finished_at = datetime.now(timezone.utc)
        await db.commit()

        # 释放机器人
        for node_id, info in state["allocation"].items():
            if info.get("robot_id"):
                robot = await db.get(Robot, info["robot_id"])
                if robot:
                    robot.status = "idle"
                    robot.current_exec_id = None
                    robot.total_tasks += 1
        await db.commit()

    await event_bus.publish(f"execution:{state['execution_id']}", {
        "type": "execution_complete" if status == "done" else "execution_stopped",
        "execution_id": state["execution_id"],
        "done": state["done_count"],
        "total": state["total_count"],
    })

    return {"messages": [{"role": "system", "content": f"执行{status}"}]}

5.3 机器人子代理

src/agents/executor/bot_agent.py
from langchain.agents import create_agent
from langchain_core.tools import BaseTool

from src.core.model_gateway import ModelGateway
from src.core.mcp_registry import MCPRegistry
from src.core.skill_runtime import SkillRuntime
from src.db.session import async_session_factory
from src.models.model_config import ModelConfig
from src.models.mcp_server import TaskMCPBinding, MCPServer
from src.models.skill import TaskSkillBinding, Skill
from sqlalchemy import select


async def create_bot_agent(robot, task):
    """为单个机器人创建子代理"""
    tools: list[BaseTool] = []

    async with async_session_factory() as db:
        # 1. 加载模型配置
        model = await db.get(ModelConfig, task.model_config_id)
        if not model:
            raise ValueError(f"Model config {task.model_config_id} not found")

        # 2. 加载 MCP 工具
        mcp_bindings = (await db.execute(
            select(TaskMCPBinding).where(
                TaskMCPBinding.task_id == task.id,
                TaskMCPBinding.is_enabled == True,
            )
        )).scalars().all()

        registry = MCPRegistry()
        for b in mcp_bindings:
            mcp = await db.get(MCPServer, b.mcp_server_id)
            mcp_tools = await registry.discover_tools(mcp)
            tools.extend(mcp_tools)

        # 3. 加载 Skills
        skill_bindings = (await db.execute(
            select(TaskSkillBinding).where(
                TaskSkillBinding.task_id == task.id,
                TaskSkillBinding.is_enabled == True,
            )
        )).scalars().all()

        for b in skill_bindings:
            skill = await db.get(Skill, b.skill_id)
            tools.append(SkillRuntime.to_langchain_tool(skill))

    # 4. 构建系统提示
    role = await _load_role(robot.role_id)
    system_prompt = f"""
你是 {robot.name}，角色是 {role.name}。
能力：{role.ability or '通用'}
{role.system_prompt or ''}
{task.system_prompt or ''}

你的任务是：
{task.prompt or task.name}

请使用可用工具完成任务，输出格式为 {task.output_format}。
"""

    # 5. 创建 Agent
    agent = create_agent(
        model=model.model_id,
        tools=tools,
        system_prompt=system_prompt,
        temperature=float(model.temperature),
        max_tokens=model.max_tokens,
        timeout=task.timeout_sec,
        max_retries=task.max_retries,
    )

    return agent


async def _load_role(role_id):
    from src.models.role import Role
    async with async_session_factory() as db:
        return await db.get(Role, role_id)
